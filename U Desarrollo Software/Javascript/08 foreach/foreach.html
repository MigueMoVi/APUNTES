forEach()

Es un método usado para iterar en los elementos de un array y aplicar una función específica (callback) para cada elemento

array.forEach(callback)

En otras palabras, ejecuta la función indicada una vez por cada elemento del array. Por ejemplo: 

let numbers = [1, 2, 3, 4, 5];

// Creamos una función para mostrar los elementos
function display(elem){
	console.log(elem)
}

// con forEach
numbers.forEach(display);

Obtenemos los elementos del arreglo en la consola


Sintaxis 

arr.forEach(function callback(currentValue, index, array) {
    // tu iterador
}[, thisArg]);

Parámetros

callback

Función a ejecutar por cada elemento, que recibe tres argumentos:

  • currentValue : El elemento actual siendo procesado en el array.
  • index : El índice del elemento actual siendo procesado en el array..
  • array : El vector en el que forEach() esta siendo aplicado.

Por ejemplo

function logArrayElements(element, index, array) {
  console.log("a[" + index + "] = " + element);
}
	
// Nótese que se evita el 2° índice ya que no hay ningún elemento en esa posición del array
[2, 5, , 9].forEach(logArrayElements);
	// salida:
	// a[0] = 2
	// a[1] = 5
	// a[2] = 9

También podríamos hacer una función para editar los elementos del arreglo y sumarles 1 a cada uno: 

let arr = [1,2,3,4,5]

arr.forEach((elem, index, originalArr) => {
		originalArr[index] = elem + 1; // Modifica directamente la posición
});

console.log(arr);

// La salida es: [2,4,6,8,10]
